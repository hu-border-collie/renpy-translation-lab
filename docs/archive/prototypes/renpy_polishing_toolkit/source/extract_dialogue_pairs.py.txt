import os
import re
import argparse
import glob

def extract_dialogue(file_path):
    extracted_pairs = []
    
    with open(file_path, 'r', encoding='utf-8') as f:
        lines = f.readlines()
        
    current_english = None
    current_label = None
    
    # Regex to capture dialogue lines
    # Matches: # char "text" OR # "text"
    english_pattern = re.compile(r'^\s*#\s*(?:[\w\s]+\s+)?(".*")\s*$')
    # Matches: char "text" OR "text"
    chinese_pattern = re.compile(r'^\s*(?:[\w\s]+\s+)?(".*")\s*$')
    # Matches label to track context (optional)
    label_pattern = re.compile(r'^\s*translate\s+schinese\s+([\w_]+):')

    for i, line in enumerate(lines):
        line = line.strip()
        
        # Track label for context
        label_match = label_pattern.match(line)
        if label_match:
            current_label = label_match.group(1)
            continue

        # Look for English comment line
        eng_match = english_pattern.match(line)
        if eng_match:
            current_english = eng_match.group(1)
            continue
            
        # Look for Chinese translation line
        # It usually follows the English line, maybe with empty lines in between
        if current_english:
            chn_match = chinese_pattern.match(line)
            if chn_match:
                chinese_text = chn_match.group(1)
                # Store the pair
                extracted_pairs.append({
                    'file': os.path.basename(file_path),
                    'line_num': i + 1,
                    'label': current_label,
                    'english': current_english,
                    'chinese': chinese_text
                })
                current_english = None # Reset
            elif line.startswith('translate') or line.startswith('# game/'):
                 # We moved past a block without finding a matching string (or it was a different kind of comment)
                 current_english = None

    return extracted_pairs

def main():
    parser = argparse.ArgumentParser(description="Extract dialogue pairs from Ren'Py translation files.")
    parser.add_argument('-i', '--input', nargs='+', required=True, help='Input .rpy files or directories containing .rpy files')
    parser.add_argument('-o', '--output', required=True, help='Output text file path')
    args = parser.parse_args()

    # Gather target files
    target_files = []
    for path in args.input:
        if os.path.isfile(path) and path.endswith('.rpy'):
            target_files.append(path)
        elif os.path.isdir(path):
            # Recursively find .rpy files
            for root, _, files in os.walk(path):
                for file in files:
                    if file.endswith('.rpy'):
                        target_files.append(os.path.join(root, file))
        else:
            print(f"Warning: Invalid input path: {path}")

    target_files = list(set(target_files)) # Remove duplicates

    all_pairs = []
    print(f"Extracting dialogue from {len(target_files)} files...")
    
    for file_path in target_files:
        print(f"Processing {file_path}...")
        pairs = extract_dialogue(file_path)
        all_pairs.extend(pairs)

    print(f"Total extracted text pairs: {len(all_pairs)}")
    
    # Ensure output directory exists
    os.makedirs(os.path.dirname(os.path.abspath(args.output)), exist_ok=True)

    with open(args.output, 'w', encoding='utf-8') as f:
        for pair in all_pairs:
            f.write(f"[{pair['file']}:{pair['line_num']}] [{pair['label']}]\n")
            f.write(f"ENG: {pair['english']}\n")
            f.write(f"CHN: {pair['chinese']}\n")
            f.write("-" * 80 + "\n")
            
    print(f"Extraction complete. Saved to {args.output}")

if __name__ == "__main__":
    main()
