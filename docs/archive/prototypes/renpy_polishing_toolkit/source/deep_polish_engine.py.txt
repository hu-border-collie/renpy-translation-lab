import re

RENPY_TAG_PATTERN = re.compile(r'\{[^}]+\}')
BRACKET_VAR_PATTERN = re.compile(r'\[[^\]]+\]')

def shield_content(text):
    placeholders = []
    
    def tag_repl(match):
        placeholders.append(match.group(0))
        return f"__TAG_{len(placeholders)-1}__"
    
    def var_repl(match):
        placeholders.append(match.group(0))
        return f"__VAR_{len(placeholders)-1}__"
    
    text_shielded = RENPY_TAG_PATTERN.sub(tag_repl, text)
    text_shielded = BRACKET_VAR_PATTERN.sub(var_repl, text_shielded)
    return text_shielded, placeholders

def unshield_content(text, placeholders):
    for i, ph in enumerate(placeholders):
        text = text.replace(f"__TAG_{i}__", ph)
        text = text.replace(f"__VAR_{i}__", ph)
    return text

def polish_line_content(text):
    text_shielded, placeholders = shield_content(text)
    
    for pattern, repl in POLISH_REPLACEMENTS:
        try:
            text_shielded = re.sub(pattern, repl, text_shielded)
        except Exception:
            pass
            
    return unshield_content(text_shielded, placeholders)

def process_file(file_path):
    with open(file_path, 'r', encoding='utf-8') as f:
        lines = f.readlines()
        
    modified_count = 0
    new_lines = []
    
    in_translation_block = False
    for line in lines:
        stripped = line.strip()
        if stripped.startswith("translate schinese"):
            in_translation_block = True
            new_lines.append(line)
            continue
            
        if in_translation_block and not stripped.startswith("#") and stripped.startswith('"'):
            polished = polish_line_content(line)
            if polished != line:
                modified_count += 1
            new_lines.append(polished)
        elif in_translation_block and not stripped.startswith("#") and (' "' in line or ' "' in stripped):
            polished = polish_line_content(line)
            if polished != line:
                modified_count += 1
            new_lines.append(polished)
        else:
            new_lines.append(line)
            
    with open(file_path, 'w', encoding='utf-8') as f:
        f.writelines(new_lines)
        
    print(f"Processed {os.path.basename(file_path)}: Polished {modified_count} lines.")
    return modified_count

if __name__ == "__main__":
    if len(sys.argv) > 1:
        target = sys.argv[1]
        if os.path.isfile(target):
            process_file(target)
        elif os.path.isdir(target):
            for root, _, files in os.walk(target):
                for file in files:
                    if file.endswith('.rpy'):
                        process_file(os.path.join(root, file))
